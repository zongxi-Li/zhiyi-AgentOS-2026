class GuardedModelRuntime:
    async def generate_json(self, *, commit_id: str | None = None, **kwargs):
        for attempt in range(self.retries + 1):
            try:
                return await asyncio.wait_for(
                    self.delegate.generate_json(commit_id=commit_id, **kwargs),
                    timeout=kwargs["timeout_seconds"],
                )
            except StructuredGenerationError as exc:
                if exc.code not in {"MODEL_TIMEOUT", "MODEL_RATE_LIMITED", "MODEL_TEMPORARY_UNAVAILABLE"} or attempt == self.retries:
                    raise
```

`StructuredGenerationRuntime.generate_json` 增加可选 `commit_id`；`NativeGeneralAgent` 每次调用传 `context.commit_id`。最终实现还须以实例级 semaphore 和最小间隔限制并发，并将超时映射为不含正文的 `StructuredGenerationError`。

class CommunicationBroker:
    async def read_reference(self, *, run_id: str, consumer_step_id: str, output_ref: str, requested_fields: list[str], max_tokens: int, reason: str) -> ContextPack:
        rule = self.manifest.require_edge(producer_ref=output_ref, consumer_step_id=consumer_step_id)
        value = self.value_store.read_authorized(output_ref, run_id=run_id, fields=rule.allowed_fields)
        self.budget.reserve(run_id=run_id, step_id=consumer_step_id, channel=rule.channel, tokens=min(max_tokens, rule.max_tokens))

def publish(self, *, key: str, content: Mapping[str, Any]) -> bool:
    revision = sha256(canonical_json(content).encode()).hexdigest()
    if self._revisions.get(key) == revision:
        return False
    self._revisions[key] = revision
    return True
```
